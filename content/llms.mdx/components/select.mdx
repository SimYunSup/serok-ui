# Select

## Overview

A select (picker) component allows users to choose a single option from a collapsible list when space is limited. It displays the currently selected option and expands to show all available choices. Serok UI's Select component extends React Spectrum's Picker with custom styling.

## Quick Start

Minimal working example showing the most common usage pattern:

```tsx
import { Select, Item } from '@/components/ui/Select';

export function MyComponent() {
  return (
    <Select label="Choose an option">
      <Item key="option1">Option 1</Item>
      <Item key="option2">Option 2</Item>
      <Item key="option3">Option 3</Item>
    </Select>
  );
}
```

## Props

| Prop | Type | Required | Default | Description |
|------|------|----------|---------|-------------|
| `label` | `string \| ReactNode` | No | — | Select label displayed above the field |
| `placeholder` | `string` | No | — | Placeholder text when no option selected |
| `selectedKey` | `string \| number` | No | — | Controlled selected item key |
| `defaultSelectedKey` | `string \| number` | No | — | Uncontrolled initial selected key |
| `onSelectionChange` | `(key: string \| number) => void` | No | — | Called when selection changes |
| `isDisabled` | `boolean` | No | `false` | Disable user selection |
| `isLoading` | `boolean` | No | `false` | Show loading state |
| `isQuiet` | `boolean` | No | `false` | Minimal styling, border not visible |
| `isInvalid` | `boolean` | No | `false` | Validation failure state |
| `isRequired` | `boolean` | No | `false` | Mark as required field |
| `description` | `string` | No | — | Description text below the field |
| `errorMessage` | `string` | No | — | Error message for validation failures |
| `children` | `Item[]` | Yes | — | Item options as children |

**Additional Notes**:
- All props follow React Spectrum naming conventions
- Props are case-sensitive
- Single selection only (not multi-select)
- Extends all props from React Spectrum's `@react-spectrum/picker` component

## Item Props

The `Item` component (exported from Select) accepts these props:

| Prop | Type | Required | Default | Description |
|------|------|----------|---------|-------------|
| `key` | `string \| number` | Yes | — | Unique identifier for the option |
| `children` | `ReactNode` | Yes | — | Display text or content for the option |
| `isDisabled` | `boolean` | No | `false` | Disable this specific option |
| `textValue` | `string` | No | — | Text used for filtering/searching |

## Behavioral States

- **Enabled**: Normal selection possible
- **Disabled** (`isDisabled={true}`): No selection allowed, grayed out
- **Loading** (`isLoading={true}`): Shows loading state
- **Invalid** (`isInvalid={true}`): Validation error state
- **Quiet** (`isQuiet={true}`): Minimal styling, border hidden

## Examples

### Basic Usage

Simplest way to use the Select component with items.

```tsx
import { Select, Item } from '@/components/ui/Select';

export function BasicExample() {
  return (
    <Select label="Select a fruit">
      <Item key="apple">Apple</Item>
      <Item key="banana">Banana</Item>
      <Item key="orange">Orange</Item>
    </Select>
  );
}
```

### Uncontrolled Select

Select managing its own state with default selection.

```tsx
import { Select, Item } from '@/components/ui/Select';

export function UncontrolledExample() {
  return (
    <Select label="Default selection" defaultSelectedKey="option2">
      <Item key="option1">Option 1</Item>
      <Item key="option2">Option 2</Item>
      <Item key="option3">Option 3</Item>
    </Select>
  );
}
```

### Controlled Select

Select with state managed by parent component.

```tsx
import { Select, Item } from '@/components/ui/Select';
import { useState } from 'react';

export function ControlledExample() {
  const [selected, setSelected] = useState('option1');

  return (
    <Select
      label="Choose option"
      selectedKey={selected}
      onSelectionChange={setSelected}
    >
      <Item key="option1">Option 1</Item>
      <Item key="option2">Option 2</Item>
      <Item key="option3">Option 3</Item>
    </Select>
  );
}
```

### With Description

Example showing select with description and error messaging.

```tsx
import { Select, Item } from '@/components/ui/Select';
import { useState } from 'react';

export function DescriptionExample() {
  const [selected, setSelected] = useState('');
  const isInvalid = !selected;

  return (
    <Select
      label="Select an option"
      selectedKey={selected}
      onSelectionChange={setSelected}
      isRequired
      isInvalid={isInvalid}
      errorMessage={isInvalid ? 'This field is required' : ''}
      description="You must select an option"
    >
      <Item key="opt1">Option 1</Item>
      <Item key="opt2">Option 2</Item>
      <Item key="opt3">Option 3</Item>
    </Select>
  );
}
```

### Loading State

Example showing select with loading state.

```tsx
import { Select, Item } from '@/components/ui/Select';
import { useState } from 'react';

export function LoadingExample() {
  const [isLoading, setIsLoading] = useState(false);

  const handleSelectionChange = async (key: string | number) => {
    setIsLoading(true);
    try {
      await new Promise(resolve => setTimeout(resolve, 1000));
      console.log('Selected:', key);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <Select
      label="Select option"
      isLoading={isLoading}
      onSelectionChange={handleSelectionChange}
    >
      <Item key="opt1">Option 1</Item>
      <Item key="opt2">Option 2</Item>
      <Item key="opt3">Option 3</Item>
    </Select>
  );
}
```
