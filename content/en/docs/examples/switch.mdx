# Switch Example

Complete guide to using the Switch component with various configurations.

## Basic Usage

The simplest way to use the Switch component.

```tsx
import { Switch } from '@/components/ui/Switch';

export function BasicExample() {
  return <Switch>Enable notifications</Switch>;
}
```

## Uncontrolled Switch

Switch managing its own state with default initial value.

```tsx
import { Switch } from '@/components/ui/Switch';

export function UncontrolledExample() {
  return (
    <Switch defaultSelected={false}>
      Subscribe to newsletters
    </Switch>
  );
}
```

## Controlled Switch

Switch with state managed by parent component.

```tsx
import { Switch } from '@/components/ui/Switch';
import { useState } from 'react';

export function ControlledExample() {
  const [isSelected, setIsSelected] = useState(false);

  return (
    <Switch
      isSelected={isSelected}
      onChange={setIsSelected}
    >
      Dark mode
    </Switch>
  );
}
```

export function ControlledSwitchExample() {
  const [isSelected, setIsSelected] = useState(false);

  return (
    <Switch
      isSelected={isSelected}
      onChange={setIsSelected}
    >
      Dark mode {isSelected ? '(ON)' : '(OFF)'}
    </Switch>
  );
}
